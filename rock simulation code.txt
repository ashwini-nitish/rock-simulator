<script>

const heatSlider = document.getElementById("heat");
const pressureSlider = document.getElementById("pressure");
const mercury = document.getElementById("mercury");

heatSlider.oninput = function(){

    const heat = Number(this.value);

    mercury.style.height = heat + "%";

    const label = document.getElementById("heatLabel");

    if(heat < 25){

        label.innerText = "Low";
        label.style.color = "#2196F3";
        mercury.style.background = "#2196F3";
    }

    else if(heat < 50){

        label.innerText = "Warm";
        label.style.color = "#FFC107";
        mercury.style.background = "#FFC107";
    }

    else if(heat < 75){

        label.innerText = "High";
        label.style.color = "#FF9800";
        mercury.style.background = "#FF9800";
    }

    else{

        label.innerText = "Melting";
        label.style.color = "#F44336";
        mercury.style.background = "#F44336";
    }
};

pressureSlider.oninput = function(){

    document.getElementById("pressureValue").innerText =
    this.value;

};

function updateStartingRock(){

    const type =
    document.getElementById("startingRock").value;

    if(type === "igneous"){
        document.getElementById("rock").innerHTML = "🧱";
        document.getElementById("rockName").innerHTML =
        "Igneous Rock";
    }

    else{
        document.getElementById("rock").innerHTML = "🪨";
        document.getElementById("rockName").innerHTML =
        "Sedimentary Rock";
    }
}

function transformRock(){

    const heat = Number(heatSlider.value);
    const pressure = Number(pressureSlider.value);

    const startRock =
    document.getElementById("startingRock").value;

    const rock =
    document.getElementById("rock");

    const rockName =
    document.getElementById("rockName");

    const result =
    document.getElementById("result");

    const challenge =
    document.getElementById("challenge");

    if(heat >= 75){

        rock.innerHTML = "🌋";
        rockName.innerHTML = "Magma";

        result.innerHTML =
        "<h3>🔥 Rock Melted!</h3>" +
        "The heat became so high that the rock melted and turned into magma.";

        challenge.innerHTML =
        "✅ Discovery: Any rock can melt into magma.";

        return;
    }

    if(heat >= 25 && pressure >= 60){

        rock.innerHTML = "💎";
        rockName.innerHTML = "Metamorphic Rock";

        result.innerHTML =
        "<h3>✅ Metamorphic Rock Formed</h3>" +
        "Heat and pressure changed the rock without melting it.";

        challenge.innerHTML =
        "✅ Discovery: Both sedimentary and igneous rocks can become metamorphic rocks.";

        return;
    }

    if(startRock === "igneous"){

        rock.innerHTML = "🧱";
        rockName.innerHTML = "Igneous Rock";

    } else {

        rock.innerHTML = "🪨";
        rockName.innerHTML = "Sedimentary Rock";
    }

    result.innerHTML =
    "<h3>No Significant Change</h3>" +
    "The conditions were not strong enough to transform the rock.";

    challenge.innerHTML = "";
}

function resetSimulation(){

    document.getElementById("startingRock").value =
    "sedimentary";

    document.getElementById("rock").innerHTML =
    "🪨";

    document.getElementById("rockName").innerHTML =
    "Sedimentary Rock";

    heatSlider.value = 40;
    pressureSlider.value = 50;

    mercury.style.height = "40%";
    mercury.style.background = "#FFC107";

    document.getElementById("heatLabel").innerHTML =
    "Warm";

    document.getElementById("heatLabel").style.color =
    "#FFC107";

    document.getElementById("pressureValue").innerHTML =
    "50";

    document.getElementById("result").innerHTML =
    'Adjust the heat and pressure and click "Transform Rock".';

    document.getElementById("challenge").innerHTML =
    "";
}

</script>
